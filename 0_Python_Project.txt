# Пользователь вводить размеры прямоугольника, допустим в см.
print("Введите длину и ширенму прямоугольника в см.")
length, width = map(float,input().split())
# Выводим данные о периметре.
P = (length + width + length + width)
print("Периметр прямоугольника равен:", P,'см')

# Выводим площадь прямоугольника.
S = ( length * width)
print("Площадь прямоугольника равна:", S,"см")
